const {
    obtenerSesion,
    crearSesion,
    actualizarPaso,
    aumentarIntento,
    reiniciarIntentos,
    reiniciarSesion
} = require("./sesion.service");

const {
    validarNombre,
    validarCURP,
    convertirFecha
} = require("../utils/validaciones");

const {
    sexo,
    nivelEstudios,
    disponibilidad,
    respuestaSiNo,
    horario
} = require("../utils/catalogos");

const {
    generarMenu
} = require("../utils/menu");

const {
    actualizarDato,
    generarFolio,
    completarPerfil,
    obtenerBuscador
} = require("./buscador.service");

const {
    buscarVacantes
} = require("./vacante.service");

async function procesarMensaje(
    sesion,
    mensaje,
    telefono
) {
    let respuesta = "";
    let siguientePaso = sesion.paso_actual;

    mensaje = mensaje.trim();

    // comandos generales

    const comando = mensaje.toLowerCase();


    if(
        comando === "7" ||
        comando === "salir" ||
        comando === "cancelar"
    ){

        await reiniciarSesion(telefono);


        return {
            respuesta:
            "👋 Gracias por utilizar el Asistente SNE.\nHasta pronto.",
            siguientePaso:"finalizado"
        };

    }

    if(
        comando === "6" ||
        comando === "menu" ||
        comando === "inicio"
    ){

        return {
            respuesta:
`👋 Bienvenido al Asistente SNE

¿Qué deseas hacer?

1️⃣ Buscar vacantes
2️⃣ Registrar mis datos
3️⃣ Actualizar mis datos
4️⃣ Consultar mi folio
5️⃣ Hablar con un asesor
6️⃣ Menú principal
7️⃣ Salir`,
            siguientePaso:"menu"
        };

    }


    switch (sesion.paso_actual) {
        case "inicio":
            respuesta =
                `👋 Hola, soy el asistente del SNE.

¿En qué puedo ayudarte?

1️⃣ Buscar vacantes
2️⃣ Registrar mis datos
3️⃣ Actualizar mis datos
4️⃣ Consultar mi folio
5️⃣ Hablar con un asesor
6️⃣ Menú principal
7️⃣ Salir`;

            siguientePaso = "menu";
            break;

        case "menu":

            switch (mensaje) {

                case "1":

                    respuesta = `
🔎 Buscar vacantes

¿Qué puesto estás buscando?
`;

                    siguientePaso = "buscar_puesto";

                    break;


                case "2": {

                    const usuario = await obtenerBuscador(telefono);

                    if (usuario && usuario.perfil_completo) {

                        respuesta = `
✅ Ya tienes un perfil registrado.

📄 Folio: ${usuario.folio}
No es necesario volver a capturar tus datos.

¿Qué deseas hacer?

1️⃣ Buscar vacantes
2️⃣ Ver mis datos
3️⃣ Actualizar mis datos
4️⃣ Salir
`;

                        siguientePaso = "perfil_completo";

                    } else {

                        respuesta = `
Perfecto 👍

Vamos a registrar tus datos.

¿Cuál es tu nombre completo?
`;

                        siguientePaso = "esperando_nombre";
                    }

                    break;
                }


                case "3": {

                    const usuario = await obtenerBuscador(telefono);

                    if (!usuario) {

                        respuesta = `
⚠️ No encontramos ningún registro.

Selecciona la opción 2 para registrarte.
`;

                        siguientePaso = "menu";

                    } else {

                        respuesta = `
✏️ ¿Qué dato deseas actualizar?

1️⃣ Correo
2️⃣ Municipio
3️⃣ Colonia
4️⃣ Carrera
5️⃣ Experiencia
8️⃣ Puesto solicitado
9️⃣ Sueldo deseado
10️⃣ Horario
`;

                        siguientePaso = "seleccionar_actualizacion";
                    }

                    break;
                }


                case "4":

                    respuesta = `
🔎 Consultar folio.

Escribe tu folio.
`;

                    siguientePaso = "consultar_folio";

                    break;


                case "5":

                    respuesta = `
Un asesor se pondrá en contacto contigo.
`;

                    siguientePaso = "asesor";

                    break;


                case "6":

                    siguientePaso = "inicio";

                    break;


                case "7":

                    respuesta = `
👋 Gracias por utilizar el asistente SNE.
`;

                    siguientePaso = "finalizado";

                    break;


                default:
                    respuesta = `
⚠️ Selecciona una opción válida.

1️⃣ Buscar vacantes
2️⃣ Registrar mis datos
3️⃣ Actualizar mis datos
4️⃣ Consultar mi folio
5️⃣ Hablar con un asesor
6️⃣ Menú principal
7️⃣ Salir`;
                    break;
            }

            break;

        case "perfil_completo": {

            const usuario = await obtenerBuscador(telefono);

            switch (mensaje) {

                case "1":
                    siguientePaso = "buscar_puesto";
                    break;

                case "2":
                    respuesta = `
📄 Estos son tus datos registrados.

Nombre: ${usuario.nombre_completo}
CURP: ${usuario.curp}
Correo: ${usuario.correo}
Puesto: ${usuario.puesto_solicitado}
Folio: ${usuario.folio}
`;
                    break;

                case "3":
                    respuesta = `
✏️ ¿Qué dato deseas actualizar?

1️⃣ Correo
2️⃣ Municipio
3️⃣ Colonia
4️⃣ Carrera
5️⃣ Experiencia
 8 Puesto solicitado
9 Sueldo deseado
10 Horario
`;
                    siguientePaso = "seleccionar_actualizacion";
                    break;

                case "4":
                    respuesta = `
👋 Gracias por utilizar el sistema.
`;
                    siguientePaso = "finalizado";
                    break;

                default:
                    respuesta = `
⚠️ Selecciona una opción válida.

1️⃣ Buscar vacantes
2️⃣ Ver mis datos
3️⃣ Actualizar mis datos
4️⃣ Salir
`;
                    break;
            }

            break;
        }

        case "seleccionar_actualizacion":

            switch (mensaje) {
                case "1":
                    respuesta = "Escribe el nuevo correo electrónico.";
                    siguientePaso = "actualizar_correo";
                    break;

                case "2":
                    respuesta = "Escribe el nuevo municipio.";
                    siguientePaso = "actualizar_municipio";
                    break;

                case "3":
                    respuesta = "Escribe la nueva colonia.";
                    siguientePaso = "actualizar_colonia";
                    break;

                case "4":
                    respuesta = "Escribe la nueva carrera.";
                    siguientePaso = "actualizar_carrera";
                    break;

                case "5":
                    respuesta = "Escribe la experiencia laboral.";
                    siguientePaso = "actualizar_experiencia";
                    break;

                case "8":
                    respuesta = "Escribe el puesto solicitado.";
                    siguientePaso = "actualizar_puesto";
                    break;

                case "9":
                    respuesta = "Escribe el nuevo sueldo.";
                    siguientePaso = "actualizar_sueldo";
                    break;

                case "10":
                    respuesta = "Escribe el horario.";
                    siguientePaso = "actualizar_horario";
                    break;

                default:
                    respuesta = "⚠️ Selecciona una opción válida.";
                    break;
            }

            break;

        case "actualizar_correo":

            await actualizarDato(
                telefono,
                "correo",
                mensaje
            );

            respuesta = "✅ Correo actualizado correctamente.";

            siguientePaso = "menu";

            break;

        case "actualizar_municipio":

            await actualizarDato(
                telefono,
                "municipio",
                mensaje
            );

            respuesta = "✅ Municipio actualizado correctamente.";

            siguientePaso = "menu";

            break;

        case "actualizar_colonia":

            await actualizarDato(
                telefono,
                "colonia",
                mensaje
            );

            respuesta = "✅ Colonia actualizada correctamente.";

            siguientePaso = "menu";

            break;

        case "actualizar_carrera":

            await actualizarDato(
                telefono,
                "carrera",
                mensaje
            );

            respuesta = "✅ Carrera actualizada correctamente.";

            siguientePaso = "menu";

            break;

        case "actualizar_experiencia":

            await actualizarDato(
                telefono,
                "experiencia",
                mensaje
            );

            respuesta = "✅ Experiencia actualizada correctamente.";

            siguientePaso = "menu";

            break;

        case "actualizar_puesto":

            await actualizarDato(
                telefono,
                "puesto_solicitado",
                mensaje
            );

            respuesta = "✅ Puesto actualizado correctamente.";

            siguientePaso = "menu";

            break;

        case "actualizar_sueldo":

            await actualizarDato(
                telefono,
                "sueldo_deseado",
                mensaje
            );

            respuesta = "✅ Sueldo actualizado correctamente.";

            siguientePaso = "menu";

            break;

        case "actualizar_horario":

            await actualizarDato(
                telefono,
                "horario",
                mensaje
            );

            respuesta = "✅ Horario actualizado correctamente.";

            siguientePaso = "menu";

            break;

        case "esperando_nombre":
            if (!validarNombre(mensaje)) {
                respuesta =
                    "⚠️ Escribe tu nombre completo.";
            } else {
                await actualizarDato(
                    telefono,
                    "nombre_completo",
                    mensaje
                );

                respuesta =
                    `Gracias ${mensaje}.

Ahora necesito tu CURP.`;

                siguientePaso = "esperando_curp";
            }
            break;

        case "esperando_curp":
            if (!validarCURP(mensaje.toUpperCase())) {
                respuesta =
                    "⚠️ La CURP no parece válida. Intenta nuevamente.";
            } else {
                await actualizarDato(
                    telefono,
                    "curp",
                    mensaje.toUpperCase()
                );

                respuesta =
                    `✅ CURP registrada.

¿Cuál es tu fecha de nacimiento?

Ejemplo:
08 de noviembre de 2003`;

                siguientePaso = "esperando_fecha";
            }
            break;

        case "esperando_fecha": {
            const fecha = convertirFecha(mensaje);

            if (!fecha) {
                respuesta =
                    `⚠️ No pude reconocer la fecha.

Ejemplo:
08 de noviembre de 2003`;
            } else {
                await actualizarDato(
                    telefono,
                    "fecha_nacimiento",
                    fecha
                );

                respuesta =
                    `✅ Fecha registrada.

¿Cuál es tu sexo?

${generarMenu(sexo)}`;

                siguientePaso = "esperando_sexo";
            }
            break;
        }

        case "esperando_sexo":
            if (sexo[mensaje]) {
                await actualizarDato(
                    telefono,
                    "sexo",
                    sexo[mensaje]
                );

                respuesta =
                    `✅ Sexo registrado: ${sexo[mensaje]}

¿Cuál es tu municipio?`;

                siguientePaso = "esperando_municipio";
            } else {
                respuesta =
                    `⚠️ Selecciona una opción válida:

${generarMenu(sexo)}`;
            }
            break;

        case "esperando_municipio":
            if (mensaje.length < 3) {
                respuesta =
                    "⚠️ Escribe un municipio válido.";
            } else {
                await actualizarDato(
                    telefono,
                    "municipio",
                    mensaje
                );

                respuesta =
                    `✅ Municipio registrado: ${mensaje}

¿Cuál es tu nivel de estudios?

${generarMenu(nivelEstudios)}`;

                siguientePaso = "esperando_estudios";
            }
            break;

        case "esperando_estudios":
            if (nivelEstudios[mensaje]) {
                await actualizarDato(
                    telefono,
                    "nivel_estudios",
                    nivelEstudios[mensaje]
                );

                respuesta =
                    `✅ Nivel de estudios: ${nivelEstudios[mensaje]}

¿Cuál es el puesto que buscas?`;

                siguientePaso = "esperando_puesto";
            } else {
                respuesta =
                    `⚠️ Selecciona una opción válida:

${generarMenu(nivelEstudios)}`;
            }
            break;

        case "esperando_puesto":
            if (mensaje.length < 2) {
                respuesta =
                    "⚠️ Escribe un puesto válido.";
            } else {
                await actualizarDato(
                    telefono,
                    "puesto_solicitado",
                    mensaje
                );

                const folio = await generarFolio(telefono);

                respuesta =
                    `✅ Registro completado correctamente.

📄 Tu folio:
${folio}

Tu perfil básico fue creado.

Completar tu información aumenta tus posibilidades de encontrar mejores oportunidades laborales y recibir vacantes más adecuadas.

¿Qué deseas hacer?

1️⃣ Buscar vacantes
2️⃣ Completar mi perfil
3️⃣ Ver mis datos
4️⃣ Menú principal`;

                siguientePaso = "menu_principal";
            }
            break;

        case "finalizado":
            respuesta =
                `✅ El registro ya fue completado.

Gracias por utilizar el asistente del SNE.`;
            break;

        case "menu_principal":
            switch (mensaje) {
                case "1":
                    respuesta =
                        `🔎 Buscar vacantes

¿Qué puesto estás buscando?`;

                    siguientePaso = "buscar_vacante";
                    break;

                case "2":
                    respuesta =
                        `Completar tu perfil te dará más oportunidades.

Podrás agregar:

✅ Experiencia laboral
✅ Carrera
✅ Correo
✅ Colonia
✅ Licencia
✅ Vehículo
✅ Horario
✅ Sueldo esperado

¿Deseas continuar?

1️⃣ Sí
2️⃣ No`;

                    siguientePaso = "confirmar_completar_perfil";
                    break;

                case "3":
                    respuesta =
                        `📄 Consulta de información.

Escribe tu folio.

Ejemplo:

SNE-2026-000001`;

                    siguientePaso = "consultar_folio";
                    break;

                case "4":
                    respuesta =
                        `🏢 Menú principal

1️⃣ Buscar vacantes
2️⃣ Completar mi perfil
3️⃣ Ver mis datos
4️⃣ Menú principal`;

                    siguientePaso = "inicio"; 
                    break;

                default:
                    respuesta =
                        `⚠️ Selecciona una opción válida.

1️⃣ Buscar vacantes
2️⃣ Completar mi perfil
3️⃣ Ver mis datos
4️⃣ Menú principal`;
                    break;
            }
            break;

        case "confirmar_completar_perfil":
            if (mensaje === "1") {
                respuesta =
                    `Perfecto 👍

¿Cuál es tu correo electrónico?`;

                siguientePaso = "esperando_correo";
            } else if (mensaje === "2") {
                respuesta =
                    `De acuerdo.

Regresando al menú principal.

1️⃣ Buscar vacantes
2️⃣ Completar mi perfil
3️⃣ Ver mis datos
4️⃣ Menú principal`;

                siguientePaso = "menu_principal";
            } else {
                respuesta =
                    `⚠️ Selecciona una opción válida.

1️⃣ Sí
2️⃣ No`;
            }
            break;

        case "esperando_correo":
            // Validación básica de correo
            if (!mensaje.includes('@') || !mensaje.includes('.')) {
                respuesta =
                    "⚠️ Escribe un correo electrónico válido (ejemplo: usuario@dominio.com)";
            } else {
                await actualizarDato(
                    telefono,
                    "correo",
                    mensaje
                );

                respuesta =
                    `✅ Correo registrado.

¿Cuál es tu colonia?`;

                siguientePaso = "esperando_colonia";
            }
            break;

        case "esperando_colonia":
            if (mensaje.length < 3) {
                respuesta =
                    "⚠️ Escribe una colonia válida.";
            } else {
                await actualizarDato(
                    telefono,
                    "colonia",
                    mensaje
                );

                respuesta =
                    `✅ Colonia registrada.

¿Cuál es tu carrera o profesión?`;

                siguientePaso = "esperando_carrera";
            }
            break;

        case "esperando_carrera":
            if (mensaje.length < 2) {
                respuesta =
                    "⚠️ Escribe una carrera o profesión válida.";
            } else {
                await actualizarDato(
                    telefono,
                    "carrera",
                    mensaje
                );

                respuesta =
                    `✅ Carrera registrada.

Cuéntame tu experiencia laboral.`;

                siguientePaso = "esperando_experiencia";
            }
            break;

        case "esperando_experiencia":
            if (mensaje.length < 5) {
                respuesta =
                    "⚠️ Escribe una descripción más detallada de tu experiencia.";
            } else {
                await actualizarDato(
                    telefono,
                    "experiencia",
                    mensaje
                );

                respuesta =
                    `✅ Experiencia registrada.

¿Tienes licencia de conducir?

1️⃣ Sí
2️⃣ No`;

                siguientePaso = "perfil_licencia";
            }
            break;

        case "perfil_licencia":
            if (respuestaSiNo[mensaje]) {
                await actualizarDato(
                    telefono,
                    "licencia_conducir",
                    mensaje === "1"
                );

                respuesta =
                    `✅ Licencia registrada: ${respuestaSiNo[mensaje]}

¿Tienes vehículo propio?

1️⃣ Sí
2️⃣ No`;

                siguientePaso = "perfil_vehiculo";
            } else {
                respuesta =
                    `⚠️ Selecciona una opción válida.

1️⃣ Sí
2️⃣ No`;
            }
            break;

        case "perfil_vehiculo":
            if (respuestaSiNo[mensaje]) {
                await actualizarDato(
                    telefono,
                    "vehiculo_propio",
                    mensaje === "1"
                );

                respuesta =
                    `✅ Vehículo registrado: ${respuestaSiNo[mensaje]}

¿Tienes disponibilidad para viajar?

1️⃣ Sí
2️⃣ No`;

                siguientePaso = "perfil_viajar";
            } else {
                respuesta =
                    `⚠️ Selecciona una opción válida.

1️⃣ Sí
2️⃣ No`;
            }
            break;

        case "perfil_viajar":
            if (respuestaSiNo[mensaje]) {
                await actualizarDato(
                    telefono,
                    "disponibilidad_viajar",
                    mensaje === "1"
                );

                respuesta =
                    `✅ Disponibilidad registrada: ${respuestaSiNo[mensaje]}

¿Cuál es tu horario disponible?

${generarMenu(horario)}`;

                siguientePaso = "perfil_horario";
            } else {
                respuesta =
                    `⚠️ Selecciona una opción válida.

1️⃣ Sí
2️⃣ No`;
            }
            break;

        case "perfil_horario":

            if (horario[mensaje]) {

                await actualizarDato(
                    telefono,
                    "horario",
                    horario[mensaje]
                );

                respuesta =
                    `✅ Horario registrado: ${horario[mensaje]}

¿Cuál es tu sueldo esperado?

Ejemplo:
$15,000 MXN`;

                siguientePaso = "perfil_sueldo";
            } else {
                respuesta =
                    `⚠️ Selecciona una opción válida:

${generarMenu(horario)}`;
            }
            break;

        case "perfil_sueldo": {
            
            // Validación básica de sueldo
            const sueldoLimpio = mensaje.replace(/[^0-9]/g, '');
            if (sueldoLimpio.length < 3) {
                respuesta =
                    "⚠️ Escribe un sueldo válido (ejemplo: $15,000 MXN)";
            } else {
                await actualizarDato(
                    telefono,
                    "sueldo_deseado",
                    mensaje
                );
                // Marcar el perfil como completo
             await completarPerfil(telefono);

                const folio = await generarFolio(telefono);

                respuesta =
                    `🎉 ¡Perfil completado exitosamente!

📄 Tu folio: ${folio}


Tu perfil está al 100% y ahora tienes mayores oportunidades de encontrar el empleo ideal.

¿Qué deseas hacer?

1️⃣ Buscar vacantes
2️⃣ Ver mis datos
3️⃣ Menú principal`;

                siguientePaso = "menu_principal";
            }
            break;
        }

        case "buscar_puesto":

    if (mensaje.length < 2) {

        respuesta =
        `⚠️ Escribe un puesto válido para buscar.

Ejemplo:
- Chofer
- Auxiliar
- Ventas
- Administrativo`;

    } else {


        const vacantes =
        await buscarVacantes(mensaje);



        if(vacantes.length === 0){


            respuesta =
            `😕 No encontré vacantes para:

"${mensaje}"

Puedes intentar con otro puesto.

Escribe otro puesto o:

6️⃣ Menú principal
7️⃣ Salir`;


            // se queda buscando
            siguientePaso = "buscar_puesto";


        }else{


            respuesta =
            `🔎 Encontré estas vacantes:

`;



            vacantes.forEach((v,index)=>{


                respuesta +=
`
${index + 1}️⃣ ${v.puesto}

🏢 Empresa:
${v.empresa}

📍 Ubicación:
${v.municipio}, ${v.entidad}

💰 Salario:
$${v.salario}

📚 Experiencia:
${v.experiencia}

`;

            });



            respuesta +=
`
¿Qué deseas hacer?

1️⃣ Ver detalle de una vacante
2️⃣ Nueva búsqueda
6️⃣ Menú principal
7️⃣ Salir`;



            siguientePaso = "seleccionar_vacante";

        }

    }

break;

        case "seleccionar_vacante":

            if (mensaje === "2") {

                respuesta =
                    `🔎 Nueva búsqueda

¿Qué puesto estás buscando?`;

                siguientePaso = "buscar_puesto";

            } else if (!isNaN(mensaje)) {

             
                respuesta =
                    `📋 Mostrando detalle de la vacante seleccionada.

¿Qué deseas hacer?

1️⃣ Postularme
2️⃣ Buscar otra vacante
6️⃣ Menú principal
7️⃣ Salir`;

                siguientePaso = "detalle_vacante";

            } else {

                respuesta =
                    `⚠️ Selecciona una opción válida.`;

            }

        break;

        case "detalle_vacante":
            switch (mensaje) {
                case "1":
                    respuesta =
                        `✅ ¡Postulación exitosa!

Tu solicitud ha sido enviada. El reclutador se pondrá en contacto contigo.

¿Qué deseas hacer?

1️⃣ Buscar más vacantes
2️⃣ Menú principal`;

                    siguientePaso = "menu_principal";
                    break;

                case "2":
                    respuesta =
                        `🔎 Nueva búsqueda

                    siguientePaso = "buscar_puesto";
                    break;

                case "3":
                    respuesta =
                        `🏢 Menú principal

1️⃣ Buscar vacantes
2️⃣ Completar mi perfil
3️⃣ Ver mis datos
4️⃣ Menú principal`;

                    siguientePaso = "menu_principal";
                    break;

                default:
                    respuesta =
                        `⚠️ Selecciona una opción válida.

1️⃣ Postularme
2️⃣ Buscar otra vacante
3️⃣ Menú principal`;
                    break;
            }
            break;

        case "consultar_folio":
            // Validación básica de folio
            if (mensaje.length < 5) {
                respuesta =
                    "⚠️ Escribe un folio válido (ejemplo: SNE-2026-000001)";
            } else {
                // Aquí iría la lógica de consulta de folio
                respuesta =
                    `📄 Información del folio: ${mensaje}

Nombre: Juan Pérez
CURP: PEREJ123456
Estado: Activo
Perfil: Completado al 80%

¿Qué deseas hacer?

1️⃣ Completar mi perfil
2️⃣ Buscar vacantes
3️⃣ Menú principal`;

                siguientePaso = "menu_principal";
            }
            break;

        default:
            respuesta =
                `No entendí tu mensaje.

Por favor, selecciona una opción válida.`;
            break;
    }

    return {
        respuesta,
        siguientePaso
    };
}

async function verificarPerfilCompleto(telefono) {

    const resultado = await db.query(
        `
        SELECT *
        FROM buscadores_empleo
        WHERE telefono = $1
        `,
        [telefono]
    );

    if (resultado.rows.length === 0) {
        return false;
    }

    const usuario = resultado.rows[0];

    const completo = (
        usuario.nombre_completo !== null &&
        usuario.curp !== null &&
        usuario.fecha_nacimiento !== null &&
        usuario.sexo !== null &&
        usuario.correo !== null &&
        usuario.municipio !== null &&
        usuario.colonia !== null &&
        usuario.nivel_estudios !== null &&
        usuario.carrera !== null &&
        usuario.experiencia !== null &&
        usuario.puesto_solicitado !== null &&
        usuario.sueldo_deseado !== null &&
        usuario.horario !== null
    );

    if (completo) {

        await db.query(
            `
            UPDATE buscadores_empleo
            SET perfil_completo = TRUE
            WHERE telefono = $1
            `,
            [telefono]
        );

        console.log("Perfil actualizado");

        return true;
    }

    return false;
}

async function procesarMensaje(
    sesion,
    mensaje,
    telefono
) {

    let respuesta = "";

    let siguientePaso =
        sesion.paso_actual;


    // ==========================================
    // NORMALIZAR MENSAJE
    // ==========================================

    mensaje =
        String(mensaje ?? "").trim();


    const comando =
        mensaje.toLowerCase();


    // ==========================================
    // COMANDO: SALIR
    // ==========================================

    if (
        comando === "7" ||
        comando === "salir" ||
        comando === "cancelar"
    ) {

        await reiniciarSesion(telefono);

        return {
            respuesta:
`👋 Gracias por utilizar el Asistente SNE.

Hasta pronto.`,
            siguientePaso: "finalizado"
        };

    }


    // ==========================================
    // COMANDO: MENÚ PRINCIPAL
    // ==========================================

    if (
        comando === "6" ||
        comando === "menu" ||
        comando === "inicio"
    ) {

        return {
            respuesta:
`👋 Bienvenido al Asistente SNE

¿Qué deseas hacer?

1️⃣ Buscar vacantes
2️⃣ Registrar mis datos
3️⃣ Actualizar mis datos
4️⃣ Consultar mi folio
5️⃣ Hablar con un asesor
6️⃣ Menú principal
7️⃣ Salir`,
            siguientePaso: "menu"
        };

    }


    // ==========================================
    // SWITCH PRINCIPAL
    // ==========================================

    switch (sesion.paso_actual) {


        // ==========================================
        // INICIO
        // ==========================================

        case "inicio":

            respuesta =
`👋 Hola, soy el asistente del SNE.

¿En qué puedo ayudarte?

1️⃣ Buscar vacantes
2️⃣ Registrar mis datos
3️⃣ Actualizar mis datos
4️⃣ Consultar mi folio
5️⃣ Hablar con un asesor
6️⃣ Menú principal
7️⃣ Salir`;

            siguientePaso = "menu";

            break;


        // ==========================================
        // MENÚ
        // ==========================================

        case "menu": {

            const resultado =
                await procesarMenu(
                    mensaje,
                    telefono
                );

            respuesta =
                resultado.respuesta;

            siguientePaso =
                resultado.siguientePaso;

            break;
        }


        // ==========================================
        // REGISTRO
        // ==========================================

        case "esperando_nombre":
        case "esperando_curp":
        case "esperando_fecha":
        case "esperando_sexo":
        case "esperando_municipio":
        case "esperando_estudios":
        case "esperando_puesto": {

            const resultado =
                await procesarRegistro(
                    sesion,
                    mensaje,
                    telefono
                );

            respuesta =
                resultado.respuesta;

            siguientePaso =
                resultado.siguientePaso;

            break;
        }


        // ==========================================
        // ACTUALIZACIÓN DE DATOS
        // ==========================================

        case "perfil_completo":
        case "seleccionar_actualizacion":
        case "actualizar_correo":
        case "actualizar_municipio":
        case "actualizar_categoria_carrera":
        case "actualizar_carrera":
        case "actualizar_colonia":
        case "actualizar_experiencia":
        case "actualizar_puesto":
        case "actualizar_sueldo":
        case "actualizar_horario": {

            const resultado =
                await procesarActualizacion(
                    sesion,
                    mensaje,
                    telefono
                );

            respuesta =
                resultado.respuesta;

            siguientePaso =
                resultado.siguientePaso;

            break;
        }


        // ==========================================
        // COMPLETAR PERFIL
        // ==========================================

        case "confirmar_completar_perfil":
        case "esperando_correo":
        case "esperando_colonia":
        case "esperando_carrera":
        case "esperando_experiencia":
        case "perfil_licencia":
        case "perfil_vehiculo":
        case "perfil_viajar":
        case "perfil_horario":
        case "perfil_sueldo": {

            const resultado =
                await procesarPerfil(
                    sesion,
                    mensaje,
                    telefono
                );

            respuesta =
                resultado.respuesta;

            siguientePaso =
                resultado.siguientePaso;

            break;
        }


        // ==========================================
        // VACANTES
        // ==========================================

        case "buscar_puesto":
        case "seleccionar_vacante":
        case "detalle_vacante": {

            const resultado =
                await procesarVacantes(
                    sesion,
                    mensaje,
                    telefono
                );

            respuesta =
                resultado.respuesta;

            siguientePaso =
                resultado.siguientePaso;

            break;
        }


        // ==========================================
        // FOLIO
        // ==========================================

        case "consultar_folio":

        case "buscar_vacantes":
        case "opciones_vacantes":

                    telefono

                resultado.siguientePaso;

        }

        // FINALIZADO
        // ==========================================

        case "finalizado":

            respuesta =
`👋 Esta conversación ha finalizado.

Escribe "inicio" para volver a comenzar.`;

            siguientePaso =
                "finalizado";

            siguientePaso =
                resultado.siguientePaso;

            break;
        }


        case "consultar_folio":
        case "folio_opciones":
        case "buscar_vacantes":
        case "opciones_vacantes":
        case "perfil_ya_completo": {

            const resultado =
                await procesarFolio(
                    sesion,
                    mensaje,
                    telefono
                );

            respuesta =
                resultado.respuesta;

            siguientePaso =
                resultado.siguientePaso;

            break;
        }



        case "finalizado":

            respuesta =
`👋 Esta conversación ha finalizado.

Escribe "inicio" para volver a comenzar.`;

            siguientePaso =
                "finalizado";

            break;


        default:

            respuesta =
`⚠️ No entendí tu mensaje.

Por favor, escribe "inicio" para volver al menú principal.`;

            siguientePaso =
                "menu";

            break;

    }


    return {
        respuesta,
        siguientePaso
    };

}

module.exports = {
    procesarMensaje
};

const db = require("../config/database");


async function crearPostulacion(
    buscadorId,
    vacanteId
) {

    const resultado = await db.query(
        `
        INSERT INTO postulaciones
        (
            buscador_id,
            vacante_id
        )
        VALUES ($1, $2)
        ON CONFLICT (buscador_id, vacante_id)
        DO NOTHING
        RETURNING *
        `,
        [
            buscadorId,
            vacanteId
        ]
    );

    return resultado.rows[0] || null;
}


async function obtenerPostulacionesPorBuscador(
    buscadorId
) {

    const resultado = await db.query(
        `
        SELECT
            p.id,
            p.fecha_postulacion,
            p.estado,

            v.id AS vacante_id,
            v.folio_oferta,
            v.puesto,
            v.empresa,
            v.municipio,
            v.entidad,
            v.salario

        FROM postulaciones p

        INNER JOIN vacantes v
            ON v.id = p.vacante_id

        WHERE p.buscador_id = $1

        ORDER BY p.fecha_postulacion DESC
        `,
        [
            buscadorId
        ]
    );

    return resultado.rows;
}


module.exports = {

    obtenerPostulacionesPorBuscador
